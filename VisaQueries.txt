```sql
    @p_fileprocessid INT
AS

    SET NOCOUNT ON;

    DECLARE @v_businessdate DATE;
    DECLARE @err_context NVARCHAR(MAX);
    DECLARE @v_successcount BIGINT = 0;

    BEGIN TRY

        ------------------------------------------------------------
        -- 1. Get business date from fileuploaddata
        ------------------------------------------------------------
        SELECT
            @v_businessdate =
                dbo.usp_convertjuliandate(
                    '2025',
                    SUBSTRING(filename, 11, 3)
                )
        FROM dbo.fileuploaddata
        WHERE id = @p_fileprocessid;


        ------------------------------------------------------------
        -- 2. Check whether records already exist
        ------------------------------------------------------------
        IF NOT EXISTS
        (
            SELECT 1
            FROM dbo.Visa
            WHERE businessdate = @v_businessdate
              AND fileprocessid = @p_fileprocessid
        )
        BEGIN

            --------------------------------------------------------
            -- 3. Delete existing matching records
            -- Preserved from the original PostgreSQL function.
            --------------------------------------------------------
            DELETE FROM dbo.Visa
            WHERE businessdate = @v_businessdate
              AND fileprocessid = @p_fileprocessid;


            --------------------------------------------------------
            -- 4. Insert eligible transactions into Visa
            --------------------------------------------------------
            INSERT INTO dbo.Visa
            (
                FMID,
                fileProcessid,
                bankid,
                Date1,
                Req_Amount,
                Terminal_ID,

            SELECT
```

```sql
CREATE OR ALTER FUNCTION dbo.usp_isdatewithformat
(
    @p_date       VARCHAR(100),
    @p_dateformat VARCHAR(50)
)
RETURNS BIT
AS
BEGIN
    DECLARE @result BIT = 0;
    DECLARE @format_code INT;

    -- Map PostgreSQL-style format strings to SQL Server styles
    SET @format_code =
        CASE LOWER(@p_dateformat)
            WHEN 'yyyymmdd' THEN 112
            WHEN 'yyyy-mm-dd' THEN 23
            WHEN 'dd/mm/yyyy' THEN 103
            WHEN 'mm/dd/yyyy' THEN 101
            WHEN 'dd-mm-yyyy' THEN 105
            WHEN 'yyyy/mm/dd' THEN 111
            WHEN 'yyyymm' THEN 112
            ELSE NULL
        END;

    IF @p_date IS NOT NULL AND @format_code IS NOT NULL
       AND TRY_CONVERT(DATE, @p_date, @format_code) IS NOT NULL
    BEGIN
        SET @result = 1;
    END;

    RETURN @result;
END;
GO
```

```sql
CREATE OR ALTER FUNCTION dbo.usp_getvisadate
(
    @p_mmdd VARCHAR(4)
)
RETURNS DATE
AS
BEGIN
    DECLARE @finaldateQuery VARCHAR(8) = '';
    DECLARE @v_todaydate VARCHAR(4);
    DECLARE @v_todayyear VARCHAR(4);
    DECLARE @result DATE;

    SET @v_todaydate = FORMAT(GETDATE(), 'MMdd');
    SET @v_todayyear = FORMAT(GETDATE(), 'yyyy');

    IF CAST(@v_todaydate AS INT) < CAST(@p_mmdd AS INT)
    BEGIN
        SET @finaldateQuery =
            CAST(CAST(@v_todayyear AS INT) - 1 AS VARCHAR(4)) + @p_mmdd;
    END
    ELSE
    BEGIN
        SET @finaldateQuery = @v_todayyear + @p_mmdd;
    END;

    SET @result = TRY_CONVERT(DATE, @finaldateQuery, 112);

    RETURN @result;
END;
GO
```
